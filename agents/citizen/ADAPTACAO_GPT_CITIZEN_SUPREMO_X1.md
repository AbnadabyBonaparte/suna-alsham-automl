const gdprCompliance = {
  // Direito de acesso
  async getPersonalData(userId) {
    const data = await db.query(`
      SELECT * FROM user_data 
      WHERE user_id = $1
    `, [userId]);
    
    return {
      personal_info: data.personal_info,
      activity_log: data.activity_log,
      preferences: data.preferences,
      exported_at: new Date().toISOString()
    };
  },
  
  // Direito de retificação
  async updatePersonalData(userId, updates) {
    await db.query(`
      UPDATE user_data 
      SET data = data || $2, updated_at = NOW()
      WHERE user_id = $1
    `, [userId, updates]);
    
    await this.logDataChange(userId, 'update', updates);
  },
  
  // Direito ao esquecimento
  async deletePersonalData(userId) {
    await db.transaction(async (trx) => {
      // Anonimizar dados em vez de deletar para manter integridade
      await trx.query(`
        UPDATE user_data 
        SET 
          email = 'deleted_' || id || '@example.com',
          name = 'Deleted User',
          phone = NULL,
          address = NULL,
          deleted_at = NOW()
        WHERE user_id = $1
      `, [userId]);
      
      await this.logDataChange(userId, 'delete', null);
    });
  },
  
  // Log de mudanças para auditoria
  async logDataChange(userId, operation, data) {
    await db.query(`
      INSERT INTO gdpr_audit_log (
        user_id, operation, data_changed, timestamp
      ) VALUES ($1, $2, $3, NOW())
    `, [userId, operation, data]);
  }
};
