class CodexIntelligentCache:
    """
    Sistema de cache inteligente para CODEX X.1
    """
    
    def __init__(self):
        self.cache_storage = {}
        self.cache_metadata = {}
        self.similarity_threshold = 0.85
    
    def get_cached_response(self, query):
        """
        Busca resposta em cache baseada em similaridade
        """
        query_embedding = self.generate_embedding(query)
        
        for cached_query, cached_response in self.cache_storage.items():
            cached_embedding = self.cache_metadata[cached_query]['embedding']
            
            similarity = self.calculate_similarity(
                query_embedding, cached_embedding
            )
            
            if similarity > self.similarity_threshold:
                return self.adapt_cached_response(cached_response, query)
        
        return None
    
    def cache_response(self, query, response):
        """
        Armazena resposta em cache com metadados
        """
        self.cache_storage[query] = response
        self.cache_metadata[query] = {
            'embedding': self.generate_embedding(query),
            'timestamp': datetime.utcnow(),
            'usage_count': 0,
            'quality_score': self.assess_response_quality(response)
